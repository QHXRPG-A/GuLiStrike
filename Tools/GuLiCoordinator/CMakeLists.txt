cmake_minimum_required(VERSION 3.24)
project(GuLiCoordinator VERSION 0.1.0 LANGUAGES CXX)
option(GULI_COORDINATOR_EOS "Enable EOS Connect identity verification" OFF)
set(EOSSDK_ROOT "" CACHE PATH "Standalone EOS SDK root containing Include and Lib")
find_package(Boost 1.85 CONFIG REQUIRED COMPONENTS asio beast json)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_subdirectory(../../Source/GuLiNetCore GuLiNetCore)
include(CTest)
if(BUILD_TESTING)
    add_executable(GuLiNetCoreAcceptance tests/ClientAcceptance.cpp)
    target_compile_features(GuLiNetCoreAcceptance PRIVATE cxx_std_20)
    target_link_libraries(GuLiNetCoreAcceptance PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,GuLiNetCore>")
    add_test(NAME GuLiNetCoreAcceptance COMMAND GuLiNetCoreAcceptance)
endif()
add_executable(GuLiCoordinator
    src/main.cpp src/Coordinator.cpp src/Store.cpp src/Identity.cpp src/Review.cpp)
target_compile_features(GuLiCoordinator PRIVATE cxx_std_20)
target_link_libraries(GuLiCoordinator PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,GuLiNetCore>" Boost::asio Boost::beast Boost::json SQLite::SQLite3 Threads::Threads)
target_compile_definitions(GuLiCoordinator PRIVATE BOOST_ASIO_NO_DEPRECATED GULI_COORDINATOR_EOS=$<BOOL:${GULI_COORDINATOR_EOS}>)
if(WIN32)
    target_compile_definitions(GuLiCoordinator PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(GuLiCoordinator PRIVATE ws2_32 mswsock bcrypt)
endif()
if(GULI_COORDINATOR_EOS)
    find_path(EOS_INCLUDE eos_sdk.h PATHS "${EOSSDK_ROOT}/Include" REQUIRED NO_DEFAULT_PATH)
    find_library(EOS_LIBRARY NAMES EOSSDK-Win64-Shipping EOSSDK-Linux-Shipping
        PATHS "${EOSSDK_ROOT}/Lib" "${EOSSDK_ROOT}/Bin" REQUIRED NO_DEFAULT_PATH)
    target_include_directories(GuLiCoordinator PRIVATE "${EOS_INCLUDE}")
    target_link_libraries(GuLiCoordinator PRIVATE "${EOS_LIBRARY}")
    if(WIN32)
        add_custom_command(TARGET GuLiCoordinator POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${EOSSDK_ROOT}/Bin/EOSSDK-Win64-Shipping.dll" "$<TARGET_FILE_DIR:GuLiCoordinator>")
    endif()
endif()
